# Generate and Verify Secure Passwords in Python ```python import secrets import string from datetime import datetime, timedelta import re def generate_password(length=16, include_symbols=True): """Generate a secure random password with customizable options.""" characters = string.ascii_letters + string.digits if include_symbols: characters += string.punctuation while True: password = ''.join(secrets.choice(characters) for _ in range(length)) # Ensure password meets complexity requirements if (any(c.islower() for c in password) and any(c.isupper() for c in password) and any(c.isdigit() for c in password) and (not include_symbols or any(c in string.punctuation for c in password))): return password def password_strength(password): """Evaluate and return the strength of a password.""" strength = 0 length = len(password) # Length check if length >= 12: strength += 2 elif length >= 8: strength += 1 # Complexity check checks = [ bool(re.search(r'[a-z]', password)), # lowercase bool(re.search(r'[A-Z]', password)), # uppercase bool(re.search(r'[0-9]', password)), # digits bool(re.search(r'[^a-zA-Z0-9]', password)) # symbols ] strength += sum(checks) return min(strength, 5) # Max strength of 5 if __name__ == "__main__": # Example usage password = generate_password() print(f"Generated Password: {password}") strength = password_strength(password) print(f"Password Strength: {'★' * strength}{'☆' * (5 - strength)} ({strength}/5)") ``` ## Explanation This code snippet provides two main functionalities for working with passwords: 1. **Password Generation**: - Uses Python's `secrets` module (secure for cryptographic operations) to generate random passwords - Customizable length and symbol inclusion - Ensures generated passwords contain at least one lowercase, uppercase, digit, and symbol (if enabled) - Avoids common weak password patterns 2. **Password Strength Evaluation**: - Rates passwords on a 5-point scale - Considers length (longer is better) and character diversity (uppercase, lowercase, numbers, symbols) - Provides visual feedback with star ratings ### Why It's Useful 1. Security best practices recommend using randomly generated, complex passwords for online accounts 2. Helps users create strong passwords without having to think of them manually 3. Useful for applications that need to enforce password policies 4. Can be integrated into user registration flows or password reset functionality ### How to Run 1. Simply copy the code into a Python file (e.g., `password_tools.py`) 2. Run with `python password_tools.py` to see an example 3. To use in your own code: ```python from password_tools import generate_password, password_strength my_pass = generate_password(length=12, include_symbols=False) ``` The code requires Python 3.6+ (for the `secrets` module) and has no external dependencies.
